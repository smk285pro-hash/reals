    39→
    40→interface AdminProduct {

    45→

    49→  createdAt: string; updatedAt: string
    50→  reporter: { id: string; name: string | null; email: string; avatar: string | null }

    53→

    55→  range: string
    56→  usersByRole: { role: string; _count: number }[]

    93→export default function AdminDashboard() {
    94→  const { setActiveCategory } = useAppStore()
    95→  const { status: authStatus, data: session } = useSession()
    96→  const [tab, setTab] = useState<Tab>('overview')
    97→  const [stats, setStats] = useState<Stats | null>(null)
    98→  const [users, setUsers] = useState<AdminUser[]>([])
    99→  const [products, setProducts] = useState<AdminProduct[]>([])
   100→  const [reports, setReports] = useState<ReportItem[]>([])
   101→  const [analytics, setAnalytics] = useState<AnalyticsData | null>(null)
   102→  const [applications, setApplications] = useState<SellerApplication[]>([])
   103→  const [loading, setLoading] = useState(true)
   104→  const [searchQ, setSearchQ] = useState('')
   105→  const [productFilter, setProductFilter] = useState('ALL')
   106→  const [reportFilter, setReportFilter] = useState('PENDING')
   107→  const [appFilter, setAppFilter] = useState('PENDING')
   108→  const [analyticsRange, setAnalyticsRange] = useState('30d')
   109→  const [rejectingId, setRejectingId] = useState<string | null>(null)
   110→  const [rejectNote, setRejectNote] = useState('')
   111→
   112→  const userRole = (session?.user as any)?.role
   113→
   114→  // Fetch data
   115→  const fetchStats = async () => {
   116→    try {
   117→      const res = await fetch('/api/admin/stats')
   118→      if (res.ok) setStats(await res.json())
   119→    } catch {}
   120→  }
   121→
   122→  const fetchUsers = async () => {
   123→    setLoading(true)
   124→    try {
   125→      const res = await fetch(`/api/admin/users?search=${searchQ}&limit=50`)
   126→      if (res.ok) { const data = await res.json(); setUsers(data.users || []) }
   127→    } catch {} finally { setLoading(false) }
   128→  }
   129→
   130→  const fetchProducts = async () => {
   131→    setLoading(true)
   132→    try {
   133→      const res = await fetch(`/api/admin/products?status=${productFilter}&search=${searchQ}&limit=50`)
   134→      if (res.ok) { const data = await res.json(); setProducts(data.products || []) }
   135→    } catch {} finally { setLoading(false) }
   136→  }
   137→
   138→  const fetchReports = async () => {
   139→    setLoading(true)
   140→    try {
   141→      const res = await fetch(`/api/admin/reports?status=${reportFilter}&limit=50`)
   142→      if (res.ok) { const data = await res.json(); setReports(data.reports || []) }
   143→    } catch {} finally { setLoading(false) }
   144→  }
   145→
   146→  const fetchAnalytics = async () => {
   147→    setLoading(true)
   148→    try {
   149→      const res = await fetch(`/api/admin/analytics?range=${analyticsRange}`)
   150→      if (res.ok) setAnalytics(await res.json())
   151→    } catch {} finally { setLoading(false) }
   152→  }
   153→
   154→  const fetchApplications = async () => {
   155→    setLoading(true)
   156→    try {
   157→      const res = await fetch(`/api/admin/applications?status=${appFilter}`)
   158→      if (res.ok) { const data = await res.json(); setApplications(data.applications || []) }
   159→    } catch {} finally { setLoading(false) }
   160→  }
   161→
   162→  useEffect(() => { if (userRole === 'ADMIN') fetchStats() }, [userRole])
   163→  useEffect(() => { if (tab === 'users' && userRole === 'ADMIN') fetchUsers() }, [tab, userRole])
   164→  useEffect(() => { if (tab === 'products' && userRole === 'ADMIN') fetchProducts() }, [tab, productFilter, userRole])
   165→  useEffect(() => { if (tab === 'applications' && userRole === 'ADMIN') fetchApplications() }, [tab, appFilter, userRole])
   166→  useEffect(() => { if (tab === 'reports' && userRole === 'ADMIN') fetchReports() }, [tab, reportFilter, userRole])
   167→  useEffect(() => { if (tab === 'analytics' && userRole === 'ADMIN') fetchAnalytics() }, [tab, analyticsRange, userRole])
   168→
   169→  // Auth guard
   170→  if (authStatus === 'loading') {
   171→    return <div className="flex min-h-screen items-center justify-center bg-[#0f0f0f]"><Skeleton className="h-20 w-40" /></div>
   172→  }
   173→  if (authStatus === 'unauthenticated' || userRole !== 'ADMIN') {
   174→    return (
   175→      <div className="flex min-h-screen items-center justify-center bg-[#0f0f0f]">
   176→        <div className="flex flex-col items-center gap-4 rounded-2xl border border-[#303030] bg-[#181818] p-12">
   177→          <Shield className="h-16 w-16 text-red-500" />
   178→          <h2 className="text-xl font-bold text-white">403 - Access Denied</h2>
   179→          <p className="text-sm text-[#888]">Bạn không có quyền truy cập trang này</p>
   180→          <Button className="bg-[#f5a623] text-black hover:bg-[#e09515]" onClick={() => setActiveCategory('all')}>
   181→            Về trang chủ
   182→          </Button>
   183→        </div>
   184→      </div>
   185→    )
   186→  }
   187→
   188→  // Actions
   189→  const updateRole = async (id: string, role: string) => {
   190→    try {
   191→      const res = await fetch('/api/admin/users', {
   192→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   193→        body: JSON.stringify({ id, role }),
   194→      })
   195→      if (res.ok) { toast.success(`Đã cập nhật role thành ${role}`); fetchUsers() }
   196→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   197→    } catch { toast.error('Lỗi kết nối') }
   198→  }
   199→
   200→  const deleteUser = async (id: string) => {
   201→    if (!confirm('Xóa user này? Hành động không thể hoàn tác.')) return
   202→    try {
   203→      const res = await fetch('/api/admin/users', {
   204→        method: 'DELETE', headers: { 'Content-Type': 'application/json' },
   205→        body: JSON.stringify({ id }),
   206→      })
   207→      if (res.ok) { toast.success('Đã xóa user'); fetchUsers() }
   208→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   209→    } catch { toast.error('Lỗi kết nối') }
   210→  }
   211→
   212→  const reviewProduct = async (id: string, reviewStatus: string, reviewNote?: string) => {
   213→    try {
   214→      const res = await fetch('/api/admin/products', {
   215→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   216→        body: JSON.stringify({ id, reviewStatus, reviewNote }),
   217→      })
   218→      if (res.ok) { toast.success(reviewStatus === 'APPROVED' ? 'Đã duyệt' : 'Đã từ chối'); fetchProducts(); fetchStats() }
   219→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   220→    } catch { toast.error('Lỗi kết nối') }
   221→  }
   222→
   223→  const deleteProduct = async (id: string) => {
   224→    if (!confirm('Xóa sản phẩm này?')) return
   225→    try {
   226→      const res = await fetch('/api/admin/products', {
   227→        method: 'DELETE', headers: { 'Content-Type': 'application/json' },
   228→        body: JSON.stringify({ id }),
   229→      })
   230→      if (res.ok) { toast.success('Đã xóa'); fetchProducts(); fetchStats() }
   231→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   232→    } catch { toast.error('Lỗi kết nối') }
   233→  }
   234→
   235→  const togglePublish = async (id: string, published: boolean) => {
   236→    try {
   237→      const res = await fetch('/api/admin/products', {
   238→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   239→        body: JSON.stringify({ id, published }),
   240→      })
   241→      if (res.ok) { toast.success(published ? 'Đã ẩn' : 'Đã đăng'); fetchProducts() }
   242→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   243→    } catch { toast.error('Lỗi kết nối') }
   244→  }
   245→
   246→  const handleReport = async (id: string, status: string, adminNote?: string, actionType?: string) => {
   247→    try {
   248→      const res = await fetch('/api/admin/reports', {
   249→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   250→        body: JSON.stringify({ id, status, adminNote, actionType }),
   251→      })
   252→      if (res.ok) {
   253→        toast.success(status === 'ACTIONED' ? 'Đã xử lý' : status === 'DISMISSED' ? 'Đã bỏ qua' : 'Đã đánh dấu')
   254→        fetchReports(); fetchStats()
   255→      } else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   256→    } catch { toast.error('Lỗi kết nối') }
   257→  }
   258→
   259→  const reviewApplication = async (id: string, status: string, adminNote?: string) => {
   260→    try {
   261→      const res = await fetch('/api/admin/applications', {
   262→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   263→        body: JSON.stringify({ id, status, adminNote }),
   264→      })
   265→      if (res.ok) {
   266→        toast.success(status === 'APPROVED' ? 'Đã duyệt seller' : 'Đã từ chối')
   267→        fetchApplications(); fetchStats()
   268→        setRejectingId(null); setRejectNote('')
   269→      } else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   270→    } catch { toast.error('Lỗi kết nối') }
   271→  }
   272→
   273→  const roleColor: Record<string, string> = {
   274→    ADMIN: 'bg-red-500/20 text-red-400',
   275→    SELLER: 'bg-[#f5a623]/20 text-[#f5a623]',
   276→    USER: 'bg-[#3ea6ff]/20 text-[#3ea6ff]',
   277→  }
   278→
   279→  const reviewColor: Record<string, string> = {
   280→    PENDING: 'bg-yellow-500/20 text-yellow-400',
   281→    APPROVED: 'bg-[#3fb950]/20 text-[#3fb950]',
   282→    REJECTED: 'bg-red-500/20 text-red-400',
   283→  }
   284→
   285→  const reportStatusColor: Record<string, string> = {
   286→    PENDING: 'bg-yellow-500/20 text-yellow-400',
   287→    REVIEWED: 'bg-[#3ea6ff]/20 text-[#3ea6ff]',
   288→    DISMISSED: 'bg-[#888]/20 text-[#888]',
   289→    ACTIONED: 'bg-[#3fb950]/20 text-[#3fb950]',
   290→  }
   291→
   292→  const reasonLabel: Record<string, string> = {
   293→    SPAM: 'Spam',
   294→    INAPPROPRIATE: 'Không phù hợp',
   295→    COPYRIGHT: 'Vi phạm bản quyền',
   296→    OTHER: 'Khác',
   297→  }
   298→
   299→  const typeLabel: Record<string, string> = {
   300→    PRODUCT: 'Sản phẩm',
   301→    USER: 'User',
   302→    REVIEW: 'Review',
   303→  }
   304→
   305→  return (
   306→    <div className="min-h-screen bg-[#0f0f0f] text-[#f1f1f1]">
   307→      {/* Header */}
   308→      <div className="border-b border-[#303030] bg-[#0f0f0f] px-4 py-4 md:px-6">
   309→        <div className="mx-auto max-w-[1200px]">
   310→          <div className="flex items-center justify-between">
   311→            <div className="flex items-center gap-3">
   312→              <Button variant="ghost" size="icon" className="text-[#aaa] hover:bg-[#272727]" onClick={() => setActiveCategory('all')}>
   313→                <ArrowLeft className="h-5 w-5" />
   314→              </Button>
   315→              <div className="flex items-center gap-2">
   316→                <Shield className="h-6 w-6 text-red-500" />
   317→                <div>
   318→                  <h1 className="text-lg font-bold text-white">Admin Dashboard</h1>
   319→                  <p className="text-xs text-[#888]">Quản trị hệ thống RealS</p>
   320→                </div>
   321→              </div>
   322→            </div>
   323→            <div className="flex gap-1 rounded-lg bg-[#1f1f1f] p-1 overflow-x-auto">
   324→              {(['overview', 'products', 'users', 'applications', 'reports', 'analytics'] as Tab[]).map(t => (
   325→                <button
   326→                  key={t}
   327→                  className={`whitespace-nowrap rounded-md px-3 py-1.5 text-sm font-medium transition-all ${
   328→                    tab === t ? 'bg-[#f5a623] text-black' : 'text-[#aaa] hover:text-white'
   329→                  }`}
   330→                  onClick={() => setTab(t)}
   331→                >
   332→                  {tabLabels[t]}
   333→                  {t === 'applications' && stats && stats.pendingApplications > 0 && (
   334→                    <span className={`ml-1.5 inline-flex h-5 min-w-[20px] items-center justify-center rounded-full px-1.5 text-xs font-bold ${
   335→                      tab === 'applications' ? 'bg-black/20 text-black' : 'bg-red-500 text-white'
   336→                    }`}>
   337→                      {stats.pendingApplications}
   338→                    </span>
   339→                  )}
   340→                </button>
   341→              ))}
   342→            </div>
   343→          </div>
   344→        </div>
   345→      </div>
   346→
   347→      <div className="mx-auto max-w-[1200px] px-4 py-6 md:px-6">
   348→        {/* Overview Tab */}
   349→        {tab === 'overview' && stats && (
   350→          <div className="space-y-6">
   351→            <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   352→              <StatCard icon={Users} label="Tổng User" value={stats.totalUsers} color="bg-[#3ea6ff]/20 text-[#3ea6ff]" />
   353→              <StatCard icon={UserCheck} label="Seller" value={stats.totalSellers} color="bg-[#f5a623]/20 text-[#f5a623]" />
   354→              <StatCard icon={Package} label="Sản phẩm" value={stats.totalProducts} color="bg-[#a855f7]/20 text-[#a855f7]" />
   355→              <StatCard icon={CheckCircle} label="Đã đăng" value={stats.publishedProducts} color="bg-[#3fb950]/20 text-[#3fb950]" />
   356→              <StatCard icon={Clock} label="Chờ duyệt" value={stats.pendingProducts} color="bg-yellow-500/20 text-yellow-400" />
   357→              <StatCard icon={DollarSign} label="Tổng giá trị" value={`$${stats.totalRevenue.toFixed(2)}`} color="bg-[#3fb950]/20 text-[#3fb950]" />
   358→              <StatCard icon={ClipboardList} label="Đơn seller chờ duyệt" value={stats.pendingApplications} color="bg-[#f5a623]/20 text-[#f5a623]" />
   359→            </div>
   360→            {stats.pendingApplications > 0 && (
   361→              <div className="flex items-center gap-3 rounded-xl border border-[#f5a623]/30 bg-[#f5a623]/5 p-4">
   362→                <ClipboardList className="h-5 w-5 text-[#f5a623]" />
   363→                <span className="text-sm text-[#f5a623]">Có <strong>{stats.pendingApplications}</strong> đơn đăng seller đang chờ duyệt</span>
   364→                <Button size="sm" className="ml-auto bg-[#f5a623] text-black hover:bg-[#e09515]" onClick={() => { setAppFilter('PENDING'); setTab('applications') }}>
   365→                  Xem ngay
   366→                </Button>
   367→              </div>
   368→            )}
   369→            {stats.pendingProducts > 0 && (
   370→              <div className="flex items-center gap-3 rounded-xl border border-yellow-500/30 bg-yellow-500/5 p-4">
   371→                <AlertCircle className="h-5 w-5 text-yellow-400" />
   372→                <span className="text-sm text-yellow-200">Có <strong>{stats.pendingProducts}</strong> sản phẩm đang chờ duyệt</span>
   373→                <Button size="sm" className="ml-auto bg-yellow-500 text-black hover:bg-yellow-600" onClick={() => { setProductFilter('PENDING'); setTab('products') }}>
   374→                  Xem ngay
   375→                </Button>
   376→              </div>
   377→            )}
   378→          </div>
   379→        )}
   380→
   381→        {/* Products Tab */}
   382→        {tab === 'products' && (
   383→          <div className="space-y-4">
   384→            <div className="flex flex-wrap items-center gap-3">
   385→              <div className="flex items-center gap-2 rounded-lg bg-[#1f1f1f] px-3 py-2">
   386→                <Search className="h-4 w-4 text-[#888]" />
   387→                <input value={searchQ} onChange={e => setSearchQ(e.target.value)} placeholder="Tìm sản phẩm..." className="bg-transparent text-sm text-white outline-none placeholder:text-[#666]" />
   388→              </div>
   389→              <div className="flex gap-1 rounded-lg bg-[#1f1f1f] p-1">
   390→                {['ALL', 'PENDING', 'APPROVED', 'REJECTED'].map(f => (
   391→                  <button key={f} className={`rounded-md px-3 py-1 text-xs font-medium ${productFilter === f ? 'bg-[#f5a623] text-black' : 'text-[#aaa] hover:text-white'}`} onClick={() => setProductFilter(f)}>
   392→                    {f === 'ALL' ? 'Tất cả' : f === 'PENDING' ? 'Chờ duyệt' : f === 'APPROVED' ? 'Đã duyệt' : 'Từ chối'}
   393→                  </button>

   395→              </div>
   396→            </div>
   397→            {loading ? (
   398→              <div className="space-y-2">{Array.from({length:5}).map((_,i) => <Skeleton key={i} className="h-16 w-full rounded-lg bg-[#1f1f1f]" />)}</div>
   399→            ) : products.length === 0 ? (
   400→              <div className="py-20 text-center text-[#888]">Không có sản phẩm</div>
   401→            ) : (
   402→              <div className="space-y-2">
   403→                {products.map(p => (
   404→                  <div key={p.id} className="flex items-center gap-3 rounded-xl border border-[#303030] bg-[#1a1a1a] p-4">
   405→                    <div className="flex-1 space-y-1">
   406→                      <div className="flex items-center gap-2">
   407→                        <h3 className="text-sm font-medium text-[#f1f1f1]">{p.title}</h3>

   413→                        <span>•</span>

   415→                        <span>•</span>
   416→                        <span>{new Date(p.createdAt).toLocaleDateString('vi-VN')}</span>

   418→                      {p.reviewNote && <p className="text-xs text-red-400">Lý do: {p.reviewNote}</p>}
   419→                    </div>
   420→                    <div className="flex items-center gap-1">
   421→                      {p.reviewStatus === 'PENDING' && (
   422→                        <>
   423→                          <Button size="icon" variant="ghost" className="h-8 w-8 text-[#3fb950] hover:bg-[#3fb950]/10" onClick={() => reviewProduct(p.id, 'APPROVED')} title="Duyệt">
