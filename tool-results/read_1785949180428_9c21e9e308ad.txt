   105→export function SellerDashboard() {
   106→  const { setActiveCategory } = useAppStore()
   107→  const { status: authStatus } = useSession()
   108→  const [products, setProducts] = useState<Product[]>([])
   109→  const [categories, setCategories] = useState<Category[]>([])
   110→  const [loading, setLoading] = useState(true)
   111→  const [searchQ, setSearchQ] = useState('')
   112→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('ALL')
   113→  const [showForm, setShowForm] = useState(false)
   114→  const [editingId, setEditingId] = useState<string | null>(null)
   115→  const [form, setForm] = useState<ProductFormData>(emptyForm)
   116→  const [saving, setSaving] = useState(false)
   117→  const [resubmittingId, setResubmittingId] = useState<string | null>(null)
   118→  const [deleteConfirm, setDeleteConfirm] = useState<string | null>(null)
   119→
   120→  const fetchProducts = useCallback(async () => {
   121→    setLoading(true)
   122→    try {
   123→      const res = await fetch('/api/seller')
   124→      if (res.ok) {
   125→        const data = await res.json()
   126→        setProducts(data.products || [])
   127→        setCategories(data.categories || [])
   128→      } else {
   129→        const err = await res.json().catch(() => ({}))
   130→        console.error('[fetchProducts] Error:', res.status, err)
   131→        setProducts([])
   132→        setCategories([])
   133→        if (res.status === 401) {
   134→          toast.error('Vui lòng đăng nhập lại')
   135→        } else {
   136→          toast.error(err.error || 'Lỗi tải sản phẩm')
   137→        }
   138→      }
   139→    } catch (e) {
   140→      console.error('[fetchProducts] Catch:', e)
   141→      setProducts([])
   142→      setCategories([])
   143→      toast.error('Lỗi kết nối server')
   144→    } finally {
   145→      setLoading(false)
   146→    }
   147→  }, [])
   148→
   149→  useEffect(() => { fetchProducts() }, [fetchProducts])
   150→
   151→  // Auth guard
   152→  if (authStatus === 'unauthenticated') {
   153→    return <LoginGuard />
   154→  }
   155→
   156→  // Derive counts
   157→  const pendingCount = products.filter((p) => p.reviewStatus === 'PENDING').length
   158→  const approvedCount = products.filter((p) => p.reviewStatus === 'APPROVED').length
   159→  const rejectedCount = products.filter((p) => p.reviewStatus === 'REJECTED').length
   160→
   161→  // Filtered products: search + status filter
   162→  const filtered = products.filter((p) => {
   163→    const matchesSearch =
   164→      p.title.toLowerCase().includes(searchQ.toLowerCase()) ||
   165→      p.format.toLowerCase().includes(searchQ.toLowerCase())
   166→    const matchesStatus = statusFilter === 'ALL' || p.reviewStatus === statusFilter
   167→    return matchesSearch && matchesStatus
   168→  })
   169→
   170→  // Toggle publish
   171→  const togglePublish = async (product: Product) => {
   172→    try {
   173→      const res = await fetch(`/api/products/${product.id}`, {
   174→        method: 'PUT',
   175→        headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({ ...product, published: !product.published }),
   177→      })
   178→      if (res.ok) {
   179→        toast.success(product.published ? 'Đã ẩn sản phẩm' : 'Đã đăng sản phẩm')
   180→        await fetchProducts()
   181→      } else {
   182→        const err = await res.json().catch(() => ({}))
   183→        toast.error(err.error || `Lỗi cập nhật (${res.status})`)
   184→      }
   185→    } catch {
   186→      toast.error('Lỗi kết nối server')
   187→    }
   188→  }
   189→
   190→  // Delete product
   191→  const deleteProduct = async (id: string) => {
   192→    try {
   193→      const res = await fetch(`/api/products/${id}`, { method: 'DELETE' })
   194→      if (res.ok) {
   195→        toast.success('Đã xóa sản phẩm')
   196→        setDeleteConfirm(null)
   197→        await fetchProducts()
   198→      } else {
   199→        const err = await res.json().catch(() => ({}))
   200→        toast.error(err.error || `Lỗi xóa sản phẩm (${res.status})`)
   201→      }
   202→    } catch {
   203→      toast.error('Lỗi kết nối server')
   204→    }
   205→  }
   206→
   207→  // Re-submit a rejected product for review
   208→  const handleResubmit = async (id: string) => {
   209→    setResubmittingId(id)
   210→    try {
   211→      const res = await fetch(`/api/products/${id}`, {
   212→        method: 'PUT',
   213→        headers: { 'Content-Type': 'application/json' },
   214→        body: JSON.stringify({ reviewStatus: 'PENDING', reviewNote: null }),
   215→      })
   216→      if (res.ok) {
   217→        toast.success('Đã gửi lại để duyệt')
   218→        await fetchProducts()
   219→      } else {
   220→        const err = await res.json().catch(() => ({}))
   221→        toast.error(err.error || `Lỗi gửi lại (${res.status})`)
   222→      }
   223→    } catch {
   224→      toast.error('Lỗi kết nối server')
   225→    } finally {
   226→      setResubmittingId(null)
   227→    }
   228→  }
   229→
   230→  // Open form for edit
   231→  const openEdit = (product: Product) => {

   233→    setForm({
   234→      title: product.title,
   235→      description: product.description,
   236→      price: product.price,
   237→      isFree: product.isFree,
   238→      format: product.format,
   239→      categorySlug: product.categorySlug,
   240→      thumbnail: product.thumbnail,
   241→      duration: product.duration || '',
   242→      tags: product.tags,
   243→      featured: product.featured,
   244→      published: product.published,
   245→    })
   246→    setShowForm(true)
   247→  }
   248→
   249→  // Open form for create
   250→  const openCreate = () => {
   251→    setEditingId(null)
   252→    setForm(emptyForm)
   253→    setShowForm(true)
   254→  }
   255→
   256→  // Save (create or update)
   257→  const handleSave = async () => {
   258→    if (!form.title.trim()) {
   259→      toast.error('Vui lòng nhập tên sản phẩm')
   260→      return
   261→    }
   262→    if (!form.categorySlug) {
   263→      toast.error('Vui lòng chọn danh mục')
   264→      return
   265→    }
   266→    setSaving(true)
   267→    try {
   268→      let success = false
   269→      if (editingId) {
   270→        const res = await fetch(`/api/products/${editingId}`, {
   271→          method: 'PUT',
   272→          headers: { 'Content-Type': 'application/json' },
   273→          body: JSON.stringify(form),
   274→        })
   275→        if (res.ok) {
   276→          toast.success('Đã cập nhật sản phẩm')
   277→          success = true
   278→        } else {
   279→          const err = await res.json().catch(() => ({}))
   280→          toast.error(err.error || `Lỗi cập nhật (${res.status})`)
   281→        }
   282→      } else {
   283→        // New products always start as PENDING for review
   284→        const res = await fetch('/api/products', {
   285→          method: 'POST',
   286→          headers: { 'Content-Type': 'application/json' },
   287→          body: JSON.stringify({ ...form, reviewStatus: 'PENDING' }),
   288→        })
   289→        if (res.ok) {
   290→          toast.success('Đã đăng sản phẩm mới — đang chờ duyệt')
   291→          success = true
   292→        } else {
   293→          const err = await res.json().catch(() => ({}))
   294→          toast.error(err.error || `Lỗi đăng sản phẩm (${res.status})`)
   295→        }
   296→      }
   297→      if (success) {
   298→        setShowForm(false)
   299→        await fetchProducts()
   300→      }
   301→    } catch (e) {
   302→      console.error('[handleSave] Error:', e)
   303→      toast.error('Lỗi kết nối server')
   304→    } finally {
   305→      setSaving(false)
   306→    }
   307→  }
   308→
   309→  const thumbnailPresets = [
   310→    'https://images.unsplash.com/photo-1598488035243-1a23a6e36919?w=640&h=360&fit=crop',
   311→    'https://images.unsplash.com/photo-1558618660-7c0c3b1a4e93?w=640&h=360&fit=crop',
   312→    'https://images.unsplash.com/photo-1511379928520-ba4c0e00a8db?w=640&h=360&fit=crop',
   313→    'https://images.unsplash.com/photo-1516289587443-44c3f05e5c4f?w=640&h=360&fit=crop',
   314→    'https://images.unsplash.com/photo-1584900501285-24ab11a14c6c?w=640&h=360&fit=crop',
   315→    'https://images.unsplash.com/photo-1493225452364-bab4a9fcd274?w=640&h=360&fit=crop',
   316→  ]
   317→
   318→  const statusFilterOptions: { value: StatusFilter; label: string; count: number }[] = [
   319→    { value: 'ALL', label: 'Tất cả', count: products.length },
   320→    { value: 'PENDING', label: 'Chờ duyệt', count: pendingCount },
   321→    { value: 'APPROVED', label: 'Đã duyệt', count: approvedCount },
   322→    { value: 'REJECTED', label: 'Bị từ chối', count: rejectedCount },
   323→  ]
   324→
   325→  return (
   326→    <div className="min-h-screen bg-[#0f0f0f] text-[#f1f1f1]">
   327→      {/* Header */}
   328→      <div className="border-b border-[#303030] bg-[#0f0f0f] px-4 py-4 md:px-6">
   329→        <div className="mx-auto max-w-[1200px]">
   330→          <div className="flex items-center justify-between">
   331→            <div className="flex items-center gap-3">
   332→              <Button
   333→                variant="ghost"
   334→                size="icon"
   335→                className="text-[#aaa] hover:bg-[#272727] hover:text-white"
   336→                onClick={() => setActiveCategory('all')}
   337→              >
   338→                <ArrowLeft className="h-5 w-5" />
   339→              </Button>
   340→              <div>
   341→                <h1 className="text-lg font-bold text-white">Seller Dashboard</h1>
   342→                <p className="text-xs text-[#888]">Quản lý sản phẩm của bạn</p>
   343→              </div>
   344→            </div>
   345→            <div className="flex items-center gap-2">
   346→              <div className="hidden items-center gap-2 rounded-lg bg-[#1f1f1f] px-3 py-2 sm:flex">
   347→                <Search className="h-4 w-4 text-[#888]" />
   348→                <input
   349→                  value={searchQ}
   350→                  onChange={(e) => setSearchQ(e.target.value)}
   351→                  placeholder="Tìm sản phẩm..."
   352→                  className="bg-transparent text-sm text-white outline-none placeholder:text-[#666]"
   353→                />
   354→              </div>
   355→              <Button
   356→                className="gap-2 rounded-lg bg-[#f5a623] px-4 text-black hover:bg-[#e09515]"
   357→                onClick={openCreate}
   358→              >
   359→                <Plus className="h-4 w-4" />
   360→                <span className="hidden sm:inline">Đăng sản phẩm</span>
   361→              </Button>
   362→            </div>
   363→          </div>
   364→
   365→          {/* Stats bar */}
   366→          <div className="mt-4 flex flex-wrap gap-4">
   367→            <div className="rounded-lg bg-[#1f1f1f] px-4 py-2">
   368→              <span className="text-xs text-[#888]">Tổng</span>
   369→              <span className="ml-2 text-lg font-bold text-white">{products.length}</span>
   370→            </div>
   371→            <div className="rounded-lg bg-[#1f1f1f] px-4 py-2">
   372→              <span className="text-xs text-[#888]">Đăng</span>
   373→              <span className="ml-2 text-lg font-bold text-[#3fb950]">{products.filter((p) => p.published).length}</span>
   374→            </div>
   375→            <div className="rounded-lg bg-[#1f1f1f] px-4 py-2">
   376→              <span className="text-xs text-[#888]">Ẩn</span>
   377→              <span className="ml-2 text-lg font-bold text-[#ff6b6b]">{products.filter((p) => !p.published).length}</span>
   378→            </div>
   379→            <div className="rounded-lg bg-[#1f1f1f] px-4 py-2">
   380→              <span className="text-xs text-[#888]">Miễn phí</span>
   381→              <span className="ml-2 text-lg font-bold text-[#3ea6ff]">{products.filter((p) => p.isFree).length}</span>
   382→            </div>
   383→            <div className="rounded-lg bg-[#1f1f1f] px-4 py-2">
   384→              <span className="text-xs text-amber-400">Chờ duyệt</span>
   385→              <span className="ml-2 text-lg font-bold text-amber-400">{pendingCount}</span>
   386→            </div>
   387→          </div>
   388→        </div>
   389→      </div>
   390→
   391→      <div className="mx-auto max-w-[1200px] px-4 py-6 md:px-6">
   392→        {/* Pending products notice banner */}
   393→        {!loading && pendingCount > 0 && (
   394→          <div className="mb-4 flex items-center gap-3 rounded-xl border border-amber-500/30 bg-amber-500/10 px-4 py-3">
   395→            <AlertCircle className="h-5 w-5 shrink-0 text-amber-400" />
   396→            <span className="text-sm font-medium text-amber-300">
   397→              Bạn có <strong>{pendingCount}</strong> sản phẩm đang chờ duyệt
   398→            </span>
   399→          </div>
   400→        )}
   401→
   402→        {/* Status filter tabs */}
   403→        <div className="mb-4 flex flex-wrap gap-2">
   404→          {statusFilterOptions.map((opt) => (
   405→            <button
   406→              key={opt.value}
   407→              onClick={() => setStatusFilter(opt.value)}
   408→              className={`rounded-lg px-4 py-2 text-sm font-medium transition-all ${
   409→                statusFilter === opt.value
   410→                  ? 'bg-[#f5a623] text-black'
   411→                  : 'bg-[#1f1f1f] text-[#aaa] hover:bg-[#272727] hover:text-white'
   412→              }`}
   413→            >
   414→              {opt.label}
   415→              <span className={`ml-1.5 inline-flex h-5 min-w-[20px] items-center justify-center rounded-full px-1 text-xs ${
   416→                statusFilter === opt.value
   417→                  ? 'bg-black/20 text-black'
   418→                  : 'bg-[#303030] text-[#888]'
   419→              }`}>
   420→                {opt.count}
   421→              </span>
   422→            </button>
   423→          ))}
   424→        </div>
   425→
   426→        {/* Product Form Modal */}
   427→        {showForm && (
   428→          <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/70" onClick={() => setShowForm(false)}>
   429→            <div
   430→              className="mx-4 max-h-[90vh] w-full max-w-[700px] overflow-y-auto rounded-2xl border border-[#303030] bg-[#0f0f0f] p-6 shadow-2xl"
   431→              onClick={(e) => e.stopPropagation()}
   432→            >
   433→              <div className="mb-4 flex items-center justify-between">
   434→                <h2 className="text-lg font-bold text-white">
   435→                  {editingId ? 'Chỉnh sửa sản phẩm' : 'Đăng sản phẩm mới'}
   436→                </h2>
   437→                <Button variant="ghost" size="icon" className="text-[#aaa] hover:bg-[#272727]" onClick={() => setShowForm(false)}>
   438→                  <X className="h-5 w-5" />
   439→                </Button>
   440→              </div>
   441→
   442→              {/* New product review notice */}
   443→              {!editingId && (
   444→                <div className="mb-4 flex items-center gap-2 rounded-lg border border-amber-500/30 bg-amber-500/10 px-3 py-2 text-xs text-amber-300">
   445→                  <AlertCircle className="h-4 w-4 shrink-0" />
   446→                  Sản phẩm mới sẽ được gửi để duyệt trước khi hiển thị công khai.
   447→                </div>
   448→              )}
   449→
   450→              <div className="space-y-4">
   451→                <div>
   452→                  <label className="mb-1 block text-sm font-medium text-[#ccc]">Tên sản phẩm *</label>
   453→                  <Input
   454→                    value={form.title}
   455→                    onChange={(e) => setForm({ ...form, title: e.target.value })}
   456→                    placeholder="VD: TapeDrift JSFX - Saturation ấm cho Vocal"
   457→                    className="border-[#303030] bg-[#1a1a1a] text-white placeholder:text-[#666]"
   458→                  />
   459→                </div>
   460→
   461→                <div>
   462→                  <label className="mb-1 block text-sm font-medium text-[#ccc]">Mô tả</label>
   463→                  <textarea
   464→                    value={form.description}
   465→                    onChange={(e) => setForm({ ...form, description: e.target.value })}
   466→                    placeholder="Mô tả chi tiết sản phẩm..."
   467→                    rows={3}
   468→                    className="w-full rounded-lg border border-[#303030] bg-[#1a1a1a] p-3 text-sm text-white outline-none placeholder:text-[#666] focus:border-[#3ea6ff]"
   469→                  />
   470→                </div>
