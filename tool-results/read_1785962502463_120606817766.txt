    62→          ⏳ Chờ duyệt

    77→  }

   105→export function SellerDashboard() {
   106→  const { setActiveCategory, setSellerApplyModalOpen } = useAppStore()
   107→  const { status: authStatus, data: session, update: updateSession } = useSession()
   108→  const [products, setProducts] = useState<Product[]>([])
   109→  const [categories, setCategories] = useState<Category[]>([])
   110→  const [loading, setLoading] = useState(true)
   111→  const [searchQ, setSearchQ] = useState('')
   112→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('ALL')
   113→  const [showForm, setShowForm] = useState(false)
   114→  const [editingId, setEditingId] = useState<string | null>(null)
   115→  const [form, setForm] = useState<ProductFormData>(emptyForm)
   116→  const [saving, setSaving] = useState(false)
   117→  const [resubmittingId, setResubmittingId] = useState<string | null>(null)
   118→  const [deleteConfirm, setDeleteConfirm] = useState<string | null>(null)
   119→
   120→  // Verify seller status from server on mount (not just session cache)
   121→  // This catches cases where admin changed the user's seller status but session is stale
   122→  const [serverSellerStatus, setServerSellerStatus] = useState<boolean | null>(null)
   123→  useEffect(() => {
   124→    if (authStatus !== 'authenticated') return
   125→    fetch('/api/seller/apply', { method: 'GET' })
   126→      .then(r => r.ok ? r.json() : null)
   127→      .then(data => {
   128→        if (data) {
   129→          const serverIsSeller = data.isSeller === true
   130→          setServerSellerStatus(serverIsSeller)
   131→          // If server and session disagree, force refresh session to sync
   132→          const sessionIsSeller = (session?.user as any)?.isSeller === true
   133→          if (serverIsSeller !== sessionIsSeller) {
   134→            updateSession?.()
   135→          }
   136→        }
   137→      })
   138→      .catch(() => {})
   139→  }, [authStatus])
   140→
   141→  // Use server-verified status when available, otherwise fall back to session
   142→  const isVerifiedSeller = serverSellerStatus !== null ? serverSellerStatus : (session?.user as any)?.isSeller === true
   143→
   144→  const fetchProducts = useCallback(async () => {
   145→    setLoading(true)
   146→    try {
   147→      const res = await fetch('/api/seller')
   148→      if (res.ok) {
   149→        const data = await res.json()
   150→        setProducts(data.products || [])
   151→        setCategories(data.categories || [])
   152→      } else {
   153→        const err = await res.json().catch(() => ({}))
   154→        console.error('[fetchProducts] Error:', res.status, err)
   155→        setProducts([])
   156→        setCategories([])
   157→        if (res.status === 401) {
   158→          toast.error('Vui lòng đăng nhập lại')
   159→        } else {
   160→          toast.error(err.error || 'Lỗi tải sản phẩm')
   161→        }
   162→      }
   163→    } catch (e) {
   164→      console.error('[fetchProducts] Catch:', e)
   165→      setProducts([])
   166→      setCategories([])
   167→      toast.error('Lỗi kết nối server')
   168→    } finally {
   169→      setLoading(false)
   170→    }
   171→  }, [])
   172→
   173→  useEffect(() => { fetchProducts() }, [fetchProducts])
   174→
   175→  // Auth guard
   176→  if (authStatus === 'unauthenticated') {
   177→    return <LoginGuard />
   178→  }
   179→
   180→  // Seller guard - if user is not an approved seller, show registration prompt
   181→  if (authStatus === 'authenticated' && !isVerifiedSeller) {
   182→    return (
   183→      <div className="flex min-h-screen items-center justify-center bg-[#0f0f0f]">
   184→        <div className="flex flex-col items-center gap-6 rounded-2xl border border-[#303030] bg-[#181818] p-12 max-w-md text-center">
   185→          <div className="rounded-full bg-[#f5a623]/10 p-4">
   186→            <Store className="h-12 w-12 text-[#f5a623]" />
   187→          </div>
   188→          <div>
   189→            <h2 className="text-xl font-bold text-[#f1f1f1]">Bạn cần đăng ký Seller</h2>
   190→            <p className="mt-2 text-sm text-[#888]">
   191→              Để đăng và quản lý sản phẩm, bạn cần đăng ký tài khoản Seller và được admin duyệt.
   192→            </p>
   193→          </div>
   194→          <Button
   195→            className="gap-2 rounded-lg bg-[#f5a623] px-8 text-black font-semibold hover:bg-[#e09515]"
   196→            onClick={() => setSellerApplyModalOpen(true)}
   197→          >
   198→            <Store className="h-4 w-4" />
   199→            Đăng ký Seller
   200→          </Button>
   201→          <Button
   202→            variant="ghost"
   203→            className="text-[#888] hover:text-[#f1f1f1]"
   204→            onClick={() => setActiveCategory('all')}
   205→          >
   206→            ← Quay lại trang chủ
   207→          </Button>
   208→        </div>
   209→      </div>
   210→    )
   211→  }
   212→
   213→  // Derive counts
   214→  const pendingCount = products.filter((p) => p.reviewStatus === 'PENDING').length
   215→  const approvedCount = products.filter((p) => p.reviewStatus === 'APPROVED').length
   216→  const rejectedCount = products.filter((p) => p.reviewStatus === 'REJECTED').length
   217→
   218→  // Filtered products: search + status filter
   219→  const filtered = products.filter((p) => {
   220→    const matchesSearch =
   221→      p.title.toLowerCase().includes(searchQ.toLowerCase()) ||
   222→      p.format.toLowerCase().includes(searchQ.toLowerCase())
   223→    const matchesStatus = statusFilter === 'ALL' || p.reviewStatus === statusFilter
   224→    return matchesSearch && matchesStatus
   225→  })
   226→

   228→  const togglePublish = async (product: Product) => {
   229→    try {
   230→      const res = await fetch(`/api/products/${product.id}`, {
   231→        method: 'PUT',
   232→        headers: { 'Content-Type': 'application/json' },
   233→        body: JSON.stringify({ ...product, published: !product.published }),
   234→      })
   235→      if (res.ok) {
   236→        toast.success(product.published ? 'Đã ẩn sản phẩm' : 'Đã đăng sản phẩm')
   237→        await fetchProducts()
   238→      } else {
   239→        const err = await res.json().catch(() => ({}))
   240→        toast.error(err.error || `Lỗi cập nhật (${res.status})`)
   241→      }
   242→    } catch {
   243→      toast.error('Lỗi kết nối server')
   244→    }
   245→  }
   246→
   247→  // Delete product
   248→  const deleteProduct = async (id: string) => {
   249→    try {
   250→      const res = await fetch(`/api/products/${id}`, { method: 'DELETE' })
   251→      if (res.ok) {
   252→        toast.success('Đã xóa sản phẩm')
   253→        setDeleteConfirm(null)
   254→        await fetchProducts()
   255→      } else {
   256→        const err = await res.json().catch(() => ({}))
   257→        toast.error(err.error || `Lỗi xóa sản phẩm (${res.status})`)
   258→      }
   259→    } catch {
   260→      toast.error('Lỗi kết nối server')
   261→    }
   262→  }
   263→
   264→  // Re-submit a rejected product for review
   265→  const handleResubmit = async (id: string) => {
   266→    setResubmittingId(id)
   267→    try {
   268→      const res = await fetch(`/api/products/${id}`, {
   269→        method: 'PUT',
   270→        headers: { 'Content-Type': 'application/json' },
   271→        body: JSON.stringify({ reviewStatus: 'PENDING', reviewNote: null }),
   272→      })
   273→      if (res.ok) {
   274→        toast.success('Đã gửi lại để duyệt')
   275→        await fetchProducts()
   276→      } else {
   277→        const err = await res.json().catch(() => ({}))
   278→        toast.error(err.error || `Lỗi gửi lại (${res.status})`)
   279→      }
   280→    } catch {
   281→      toast.error('Lỗi kết nối server')
   282→    } finally {
   283→      setResubmittingId(null)
   284→    }
   285→  }
   286→
   287→  // Open form for edit
   288→  const openEdit = (product: Product) => {
   289→    setEditingId(product.id)
   290→    setForm({
   291→      title: product.title,
   292→      description: product.description,
   293→      price: product.price,
   294→      isFree: product.isFree,
   295→      format: product.format,
   296→      categorySlug: product.categorySlug,
   297→      thumbnail: product.thumbnail,
   298→      duration: product.duration || '',
   299→      tags: product.tags,
   300→      featured: product.featured,
   301→      published: product.published,
   302→    })
   303→    setShowForm(true)
   304→  }
   305→
   306→  // Open form for create
   307→  const openCreate = () => {
   308→    setEditingId(null)
   309→    setForm(emptyForm)
   310→    setShowForm(true)
   311→  }
   312→
   313→  // Save (create or update)
   314→  const handleSave = async () => {
   315→    if (!form.title.trim()) {
   316→      toast.error('Vui lòng nhập tên sản phẩm')
   317→      return
   318→    }
   319→    if (!form.categorySlug) {
   320→      toast.error('Vui lòng chọn danh mục')
   321→      return
   322→    }
   323→    setSaving(true)
   324→    try {
   325→      let success = false
   326→      if (editingId) {
   327→        const res = await fetch(`/api/products/${editingId}`, {
   328→          method: 'PUT',
   329→          headers: { 'Content-Type': 'application/json' },
   330→          body: JSON.stringify(form),
   331→        })
   332→        if (res.ok) {
   333→          toast.success('Đã cập nhật sản phẩm')
   334→          success = true
   335→        } else {
   336→          const err = await res.json().catch(() => ({}))
   337→          toast.error(err.error || `Lỗi cập nhật (${res.status})`)
   338→        }
   339→      } else {
   340→        // New products always start as PENDING for review
   341→        const res = await fetch('/api/products', {
   342→          method: 'POST',
   343→          headers: { 'Content-Type': 'application/json' },
   344→          body: JSON.stringify({ ...form, reviewStatus: 'PENDING' }),
   345→        })
   346→        if (res.ok) {
   347→          toast.success('Đã đăng sản phẩm mới — đang chờ duyệt')
   348→          success = true
   349→        } else {
   350→          const err = await res.json().catch(() => ({}))
   351→          toast.error(err.error || `Lỗi đăng sản phẩm (${res.status})`)
   352→        }
   353→      }
   354→      if (success) {
   355→        setShowForm(false)
   356→        await fetchProducts()
   357→      }
   358→    } catch (e) {
   359→      console.error('[handleSave] Error:', e)
   360→      toast.error('Lỗi kết nối server')
   361→    } finally {
   362→      setSaving(false)
   363→    }
   364→  }
   365→
   366→  const thumbnailPresets = [
   367→    'https://images.unsplash.com/photo-1598488035243-1a23a6e36919?w=640&h=360&fit=crop',
   368→    'https://images.unsplash.com/photo-1558618660-7c0c3b1a4e93?w=640&h=360&fit=crop',
   369→    'https://images.unsplash.com/photo-1511379928520-ba4c0e00a8db?w=640&h=360&fit=crop',
   370→    'https://images.unsplash.com/photo-1516289587443-44c3f05e5c4f?w=640&h=360&fit=crop',
   371→    'https://images.unsplash.com/photo-1584900501285-24ab11a14c6c?w=640&h=360&fit=crop',
   372→    'https://images.unsplash.com/photo-1493225452364-bab4a9fcd274?w=640&h=360&fit=crop',
   373→  ]
   374→
   375→  const statusFilterOptions: { value: StatusFilter; label: string; count: number }[] = [
   376→    { value: 'ALL', label: 'Tất cả', count: products.length },
   377→    { value: 'PENDING', label: 'Chờ duyệt', count: pendingCount },
   378→    { value: 'APPROVED', label: 'Đã duyệt', count: approvedCount },
   379→    { value: 'REJECTED', label: 'Bị từ chối', count: rejectedCount },
   380→  ]
   381→
   382→  return (
   383→    <div className="min-h-screen bg-[#0f0f0f] text-[#f1f1f1]">
   384→      {/* Header */}
   385→      <div className="border-b border-[#303030] bg-[#0f0f0f] px-4 py-4 md:px-6">
   386→        <div className="mx-auto max-w-[1200px]">
   387→          <div className="flex items-center justify-between">
   388→            <div className="flex items-center gap-3">
   389→              <Button
   390→                variant="ghost"
   391→                size="icon"
   392→                className="text-[#aaa] hover:bg-[#272727] hover:text-white"
   393→                onClick={() => setActiveCategory('all')}
   394→              >
   395→                <ArrowLeft className="h-5 w-5" />
   396→              </Button>
   397→              <div>
   398→                <h1 className="text-lg font-bold text-white">Seller Dashboard</h1>
   399→                <p className="text-xs text-[#888]">Quản lý sản phẩm của bạn</p>
   400→              </div>
   401→            </div>
   402→            <div className="flex items-center gap-2">
   403→              <div className="hidden items-center gap-2 rounded-lg bg-[#1f1f1f] px-3 py-2 sm:flex">
   404→                <Search className="h-4 w-4 text-[#888]" />
   405→                <input
   406→                  value={searchQ}
   407→                  onChange={(e) => setSearchQ(e.target.value)}
   408→                  placeholder="Tìm sản phẩm..."
   409→                  className="bg-transparent text-sm text-white outline-none placeholder:text-[#666]"
   410→                />
   411→              </div>
   412→              <Button
   413→                className="gap-2 rounded-lg bg-[#f5a623] px-4 text-black hover:bg-[#e09515]"
   414→                onClick={openCreate}
   415→              >
   416→                <Plus className="h-4 w-4" />
   417→                <span className="hidden sm:inline">Đăng sản phẩm</span>
   418→              </Button>
   419→            </div>
   420→          </div>
   421→
   422→          {/* Stats bar */}
   423→          <div className="mt-4 flex flex-wrap gap-4">
   424→            <div className="rounded-lg bg-[#1f1f1f] px-4 py-2">
   425→              <span className="text-xs text-[#888]">Tổng</span>
   426→              <span className="ml-2 text-lg font-bold text-white">{products.length}</span>
   427→            </div>
   428→            <div className="rounded-lg bg-[#1f1f1f] px-4 py-2">
   429→              <span className="text-xs text-[#888]">Đăng</span>
   430→              <span className="ml-2 text-lg font-bold text-[#3fb950]">{products.filter((p) => p.published).length}</span>
   431→            </div>
   432→            <div className="rounded-lg bg-[#1f1f1f] px-4 py-2">
   433→              <span className="text-xs text-[#888]">Ẩn</span>

   437→              <span className="text-xs text-[#888]">Miễn phí</span>

   439→            </div>
   440→            <div className="rounded-lg bg-[#1f1f1f] px-4 py-2">
   441→              <span className="text-xs text-amber-400">Chờ duyệt</span>

   443→            </div>
   444→          </div>
   445→        </div>
   446→      </div>
   447→
   448→      <div className="mx-auto max-w-[1200px] px-4 py-6 md:px-6">
   449→        {/* Pending products notice banner */}
   450→        {!loading && pendingCount > 0 && (

   452→            <AlertCircle className="h-5 w-5 shrink-0 text-amber-400" />
   453→            <span className="text-sm font-medium text-amber-300">

   455→            </span>
   456→          </div>

   458→
   459→        {/* Status filter tabs */}
   460→        <div className="mb-4 flex flex-wrap gap-2">
   461→          {statusFilterOptions.map((opt) => (

   463→              key={opt.value}
   464→              onClick={() => setStatusFilter(opt.value)}

   478→              </span>
   479→            </button>

   481→        </div>
   482→
   483→        {/* Product Form Modal */}
   484→        {showForm && (

   487→              className="mx-4 max-h-[90vh] w-full max-w-[700px] overflow-y-auto rounded-2xl border border-[#303030] bg-[#0f0f0f] p-6 shadow-2xl"
   488→              onClick={(e) => e.stopPropagation()}
   489→            >
   490→              <div className="mb-4 flex items-center justify-between">
   491→                <h2 className="text-lg font-bold text-white">
   492→                  {editingId ? 'Chỉnh sửa sản phẩm' : 'Đăng sản phẩm mới'}
