    40→

    49→  description: string | null; status: string; adminNote: string | null
    50→  createdAt: string; updatedAt: string
    51→  reporter: { id: string; name: string | null; email: string; avatar: string | null }

    54→

    94→export default function AdminDashboard() {
    95→  const { setActiveCategory } = useAppStore()
    96→  const { status: authStatus, data: session } = useSession()
    97→  const [tab, setTab] = useState<Tab>('overview')
    98→  const [stats, setStats] = useState<Stats | null>(null)
    99→  const [users, setUsers] = useState<AdminUser[]>([])
   100→  const [products, setProducts] = useState<AdminProduct[]>([])
   101→  const [reports, setReports] = useState<ReportItem[]>([])
   102→  const [analytics, setAnalytics] = useState<AnalyticsData | null>(null)
   103→  const [applications, setApplications] = useState<SellerApplication[]>([])
   104→  const [loading, setLoading] = useState(true)
   105→  const [searchQ, setSearchQ] = useState('')
   106→  const [productFilter, setProductFilter] = useState('ALL')
   107→  const [reportFilter, setReportFilter] = useState('PENDING')
   108→  const [appFilter, setAppFilter] = useState('PENDING')
   109→  const [analyticsRange, setAnalyticsRange] = useState('30d')
   110→  const [rejectingId, setRejectingId] = useState<string | null>(null)
   111→  const [rejectNote, setRejectNote] = useState('')
   112→
   113→  const userRole = (session?.user as any)?.role
   114→
   115→  // Fetch data
   116→  const fetchStats = async () => {
   117→    try {
   118→      const res = await fetch('/api/admin/stats')
   119→      if (res.ok) setStats(await res.json())
   120→    } catch {}
   121→  }
   122→
   123→  const fetchUsers = async () => {
   124→    setLoading(true)
   125→    try {
   126→      const res = await fetch(`/api/admin/users?search=${searchQ}&limit=50`)
   127→      if (res.ok) { const data = await res.json(); setUsers(data.users || []) }
   128→    } catch {} finally { setLoading(false) }
   129→  }
   130→
   131→  const fetchProducts = async () => {
   132→    setLoading(true)
   133→    try {
   134→      const res = await fetch(`/api/admin/products?status=${productFilter}&search=${searchQ}&limit=50`)
   135→      if (res.ok) { const data = await res.json(); setProducts(data.products || []) }
   136→    } catch {} finally { setLoading(false) }
   137→  }
   138→
   139→  const fetchReports = async () => {
   140→    setLoading(true)
   141→    try {
   142→      const res = await fetch(`/api/admin/reports?status=${reportFilter}&limit=50`)
   143→      if (res.ok) { const data = await res.json(); setReports(data.reports || []) }
   144→    } catch {} finally { setLoading(false) }
   145→  }
   146→
   147→  const fetchAnalytics = async () => {
   148→    setLoading(true)
   149→    try {
   150→      const res = await fetch(`/api/admin/analytics?range=${analyticsRange}`)
   151→      if (res.ok) setAnalytics(await res.json())
   152→    } catch {} finally { setLoading(false) }
   153→  }
   154→
   155→  const fetchApplications = async () => {
   156→    setLoading(true)
   157→    try {
   158→      const res = await fetch(`/api/admin/applications?status=${appFilter}`)
   159→      if (res.ok) { const data = await res.json(); setApplications(data.applications || []) }
   160→    } catch {} finally { setLoading(false) }
   161→  }
   162→
   163→  useEffect(() => { if (userRole === 'ADMIN') fetchStats() }, [userRole])
   164→  useEffect(() => { if (tab === 'users' && userRole === 'ADMIN') fetchUsers() }, [tab, userRole])
   165→  useEffect(() => { if (tab === 'products' && userRole === 'ADMIN') fetchProducts() }, [tab, productFilter, userRole])
   166→  useEffect(() => { if (tab === 'applications' && userRole === 'ADMIN') fetchApplications() }, [tab, appFilter, userRole])
   167→  useEffect(() => { if (tab === 'reports' && userRole === 'ADMIN') fetchReports() }, [tab, reportFilter, userRole])
   168→  useEffect(() => { if (tab === 'analytics' && userRole === 'ADMIN') fetchAnalytics() }, [tab, analyticsRange, userRole])
   169→
   170→  // Auth guard
   171→  if (authStatus === 'loading') {
   172→    return <div className="flex min-h-screen items-center justify-center bg-[#0f0f0f]"><Skeleton className="h-20 w-40" /></div>
   173→  }
   174→  if (authStatus === 'unauthenticated' || userRole !== 'ADMIN') {
   175→    return (
   176→      <div className="flex min-h-screen items-center justify-center bg-[#0f0f0f]">
   177→        <div className="flex flex-col items-center gap-4 rounded-2xl border border-[#303030] bg-[#181818] p-12">
   178→          <Shield className="h-16 w-16 text-red-500" />
   179→          <h2 className="text-xl font-bold text-white">403 - Access Denied</h2>
   180→          <p className="text-sm text-[#888]">Bạn không có quyền truy cập trang này</p>
   181→          <Button className="bg-[#f5a623] text-black hover:bg-[#e09515]" onClick={() => setActiveCategory('all')}>
   182→            Về trang chủ
   183→          </Button>
   184→        </div>
   185→      </div>
   186→    )
   187→  }
   188→
   189→  // Actions
   190→  const updateRole = async (id: string, role: string) => {
   191→    try {
   192→      const res = await fetch('/api/admin/users', {
   193→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   194→        body: JSON.stringify({ id, role }),
   195→      })
   196→      if (res.ok) { toast.success(`Đã cập nhật role thành ${role}`); fetchUsers() }
   197→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   198→    } catch { toast.error('Lỗi kết nối') }
   199→  }
   200→
   201→  const deleteUser = async (id: string) => {
   202→    if (!confirm('Xóa user này? Hành động không thể hoàn tác.')) return
   203→    try {
   204→      const res = await fetch('/api/admin/users', {
   205→        method: 'DELETE', headers: { 'Content-Type': 'application/json' },
   206→        body: JSON.stringify({ id }),
   207→      })
   208→      if (res.ok) { toast.success('Đã xóa user'); fetchUsers() }
   209→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   210→    } catch { toast.error('Lỗi kết nối') }
   211→  }
   212→
   213→  const reviewProduct = async (id: string, reviewStatus: string, reviewNote?: string) => {
   214→    try {
   215→      const res = await fetch('/api/admin/products', {
   216→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   217→        body: JSON.stringify({ id, reviewStatus, reviewNote }),
   218→      })
   219→      if (res.ok) { toast.success(reviewStatus === 'APPROVED' ? 'Đã duyệt' : 'Đã từ chối'); fetchProducts(); fetchStats() }
   220→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   221→    } catch { toast.error('Lỗi kết nối') }
   222→  }
   223→
   224→  const deleteProduct = async (id: string) => {
   225→    if (!confirm('Xóa sản phẩm này?')) return
   226→    try {
   227→      const res = await fetch('/api/admin/products', {
   228→        method: 'DELETE', headers: { 'Content-Type': 'application/json' },
   229→        body: JSON.stringify({ id }),
   230→      })
   231→      if (res.ok) { toast.success('Đã xóa'); fetchProducts(); fetchStats() }
   232→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   233→    } catch { toast.error('Lỗi kết nối') }
   234→  }
   235→
   236→  const togglePublish = async (id: string, published: boolean) => {
   237→    try {
   238→      const res = await fetch('/api/admin/products', {
   239→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   240→        body: JSON.stringify({ id, published }),
   241→      })
   242→      if (res.ok) { toast.success(published ? 'Đã ẩn' : 'Đã đăng'); fetchProducts() }
   243→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   244→    } catch { toast.error('Lỗi kết nối') }
   245→  }
   246→
   247→  const handleReport = async (id: string, status: string, adminNote?: string, actionType?: string) => {
   248→    try {
   249→      const res = await fetch('/api/admin/reports', {
   250→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify({ id, status, adminNote, actionType }),
   252→      })
   253→      if (res.ok) {
   254→        toast.success(status === 'ACTIONED' ? 'Đã xử lý' : status === 'DISMISSED' ? 'Đã bỏ qua' : 'Đã đánh dấu')
   255→        fetchReports(); fetchStats()
   256→      } else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   257→    } catch { toast.error('Lỗi kết nối') }
   258→  }
   259→
   260→  const reviewApplication = async (id: string, status: string, adminNote?: string) => {
   261→    try {
   262→      const res = await fetch('/api/admin/applications', {
   263→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   264→        body: JSON.stringify({ id, status, adminNote }),
   265→      })
   266→      if (res.ok) {
   267→        toast.success(status === 'APPROVED' ? 'Đã duyệt seller' : 'Đã từ chối')
   268→        fetchApplications(); fetchStats()
   269→        setRejectingId(null); setRejectNote('')
   270→      } else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   271→    } catch { toast.error('Lỗi kết nối') }
   272→  }
   273→
   274→  const roleColor: Record<string, string> = {
   275→    ADMIN: 'bg-red-500/20 text-red-400',
   276→    SELLER: 'bg-[#f5a623]/20 text-[#f5a623]',
   277→    USER: 'bg-[#3ea6ff]/20 text-[#3ea6ff]',
   278→  }
   279→
   280→  const reviewColor: Record<string, string> = {
   281→    PENDING: 'bg-yellow-500/20 text-yellow-400',
   282→    APPROVED: 'bg-[#3fb950]/20 text-[#3fb950]',
   283→    REJECTED: 'bg-red-500/20 text-red-400',
   284→  }
   285→
   286→  const reportStatusColor: Record<string, string> = {
   287→    PENDING: 'bg-yellow-500/20 text-yellow-400',
   288→    REVIEWED: 'bg-[#3ea6ff]/20 text-[#3ea6ff]',
   289→    DISMISSED: 'bg-[#888]/20 text-[#888]',
   290→    ACTIONED: 'bg-[#3fb950]/20 text-[#3fb950]',
   291→  }
   292→
   293→  const reasonLabel: Record<string, string> = {
   294→    SPAM: 'Spam',
   295→    INAPPROPRIATE: 'Không phù hợp',
   296→    COPYRIGHT: 'Vi phạm bản quyền',
   297→    OTHER: 'Khác',
   298→  }
   299→
   300→  const typeLabel: Record<string, string> = {
   301→    PRODUCT: 'Sản phẩm',
   302→    USER: 'User',
   303→    REVIEW: 'Review',
   304→  }
   305→
   306→  return (
   307→    <div className="min-h-screen bg-[#0f0f0f] text-[#f1f1f1]">
   308→      {/* Header */}
   309→      <div className="border-b border-[#303030] bg-[#0f0f0f] px-4 py-4 md:px-6">
   310→        <div className="mx-auto max-w-[1200px]">
   311→          <div className="flex items-center justify-between">
   312→            <div className="flex items-center gap-3">
   313→              <Button variant="ghost" size="icon" className="text-[#aaa] hover:bg-[#272727]" onClick={() => setActiveCategory('all')}>
   314→                <ArrowLeft className="h-5 w-5" />
   315→              </Button>
   316→              <div className="flex items-center gap-2">
   317→                <Shield className="h-6 w-6 text-red-500" />
   318→                <div>
   319→                  <h1 className="text-lg font-bold text-white">Admin Dashboard</h1>
   320→                  <p className="text-xs text-[#888]">Quản trị hệ thống RealS</p>
   321→                </div>
   322→              </div>
   323→            </div>
   324→            <div className="flex gap-1 rounded-lg bg-[#1f1f1f] p-1 overflow-x-auto">
   325→              {(['overview', 'products', 'users', 'applications', 'reports', 'analytics'] as Tab[]).map(t => (
   326→                <button
   327→                  key={t}
   328→                  className={`whitespace-nowrap rounded-md px-3 py-1.5 text-sm font-medium transition-all ${
   329→                    tab === t ? 'bg-[#f5a623] text-black' : 'text-[#aaa] hover:text-white'
   330→                  }`}
   331→                  onClick={() => setTab(t)}
   332→                >
   333→                  {tabLabels[t]}
   334→                  {t === 'applications' && stats && stats.pendingApplications > 0 && (
   335→                    <span className={`ml-1.5 inline-flex h-5 min-w-[20px] items-center justify-center rounded-full px-1.5 text-xs font-bold ${
   336→                      tab === 'applications' ? 'bg-black/20 text-black' : 'bg-red-500 text-white'
   337→                    }`}>
   338→                      {stats.pendingApplications}
   339→                    </span>
   340→                  )}
   341→                </button>
   342→              ))}
   343→            </div>
   344→          </div>
   345→        </div>
   346→      </div>
   347→
   348→      <div className="mx-auto max-w-[1200px] px-4 py-6 md:px-6">
   349→        {/* Overview Tab */}
   350→        {tab === 'overview' && stats && (
   351→          <div className="space-y-6">
   352→            <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   353→              <StatCard icon={Users} label="Tổng User" value={stats.totalUsers} color="bg-[#3ea6ff]/20 text-[#3ea6ff]" />
   354→              <StatCard icon={UserCheck} label="Seller" value={stats.totalSellers} color="bg-[#f5a623]/20 text-[#f5a623]" />
   355→              <StatCard icon={Package} label="Sản phẩm" value={stats.totalProducts} color="bg-[#a855f7]/20 text-[#a855f7]" />
   356→              <StatCard icon={CheckCircle} label="Đã đăng" value={stats.publishedProducts} color="bg-[#3fb950]/20 text-[#3fb950]" />
   357→              <StatCard icon={Clock} label="Chờ duyệt" value={stats.pendingProducts} color="bg-yellow-500/20 text-yellow-400" />
   358→              <StatCard icon={DollarSign} label="Tổng giá trị" value={`$${stats.totalRevenue.toFixed(2)}`} color="bg-[#3fb950]/20 text-[#3fb950]" />
   359→              <StatCard icon={ClipboardList} label="Đơn seller chờ duyệt" value={stats.pendingApplications} color="bg-[#f5a623]/20 text-[#f5a623]" />
   360→            </div>
   361→            {stats.pendingApplications > 0 && (
   362→              <div className="flex items-center gap-3 rounded-xl border border-[#f5a623]/30 bg-[#f5a623]/5 p-4">
   363→                <ClipboardList className="h-5 w-5 text-[#f5a623]" />
   364→                <span className="text-sm text-[#f5a623]">Có <strong>{stats.pendingApplications}</strong> đơn đăng seller đang chờ duyệt</span>
   365→                <Button size="sm" className="ml-auto bg-[#f5a623] text-black hover:bg-[#e09515]" onClick={() => { setAppFilter('PENDING'); setTab('applications') }}>
   366→                  Xem ngay
   367→                </Button>
   368→              </div>
   369→            )}
   370→            {stats.pendingProducts > 0 && (
   371→              <div className="flex items-center gap-3 rounded-xl border border-yellow-500/30 bg-yellow-500/5 p-4">
   372→                <AlertCircle className="h-5 w-5 text-yellow-400" />
   373→                <span className="text-sm text-yellow-200">Có <strong>{stats.pendingProducts}</strong> sản phẩm đang chờ duyệt</span>
   374→                <Button size="sm" className="ml-auto bg-yellow-500 text-black hover:bg-yellow-600" onClick={() => { setProductFilter('PENDING'); setTab('products') }}>
   375→                  Xem ngay
   376→                </Button>
   377→              </div>
   378→            )}
   379→          </div>
   380→        )}
   381→
   382→        {/* Products Tab */}
   383→        {tab === 'products' && (
   384→          <div className="space-y-4">
   385→            <div className="flex flex-wrap items-center gap-3">
   386→              <div className="flex items-center gap-2 rounded-lg bg-[#1f1f1f] px-3 py-2">
   387→                <Search className="h-4 w-4 text-[#888]" />
   388→                <input value={searchQ} onChange={e => setSearchQ(e.target.value)} placeholder="Tìm sản phẩm..." className="bg-transparent text-sm text-white outline-none placeholder:text-[#666]" />
   389→              </div>
   390→              <div className="flex gap-1 rounded-lg bg-[#1f1f1f] p-1">
   391→                {['ALL', 'PENDING', 'APPROVED', 'REJECTED'].map(f => (
   392→                  <button key={f} className={`rounded-md px-3 py-1 text-xs font-medium ${productFilter === f ? 'bg-[#f5a623] text-black' : 'text-[#aaa] hover:text-white'}`} onClick={() => setProductFilter(f)}>
   393→                    {f === 'ALL' ? 'Tất cả' : f === 'PENDING' ? 'Chờ duyệt' : f === 'APPROVED' ? 'Đã duyệt' : 'Từ chối'}
   394→                  </button>
   395→                ))}
   396→              </div>
   397→            </div>
   398→            {loading ? (
   399→              <div className="space-y-2">{Array.from({length:5}).map((_,i) => <Skeleton key={i} className="h-16 w-full rounded-lg bg-[#1f1f1f]" />)}</div>
   400→            ) : products.length === 0 ? (
   401→              <div className="py-20 text-center text-[#888]">Không có sản phẩm</div>
   402→            ) : (
   403→              <div className="space-y-2">
   404→                {products.map(p => (
   405→                  <div key={p.id} className="flex items-center gap-3 rounded-xl border border-[#303030] bg-[#1a1a1a] p-4">
   406→                    <div className="flex-1 space-y-1">
   407→                      <div className="flex items-center gap-2">
   408→                        <h3 className="text-sm font-medium text-[#f1f1f1]">{p.title}</h3>
   409→                        <Badge className={reviewColor[p.reviewStatus] || ''}>{p.reviewStatus}</Badge>

   411→                      </div>

   413→                        <span>{p.seller.name || p.seller.email}</span>
   414→                        <span>•</span>

   416→                        <span>•</span>
   417→                        <span>{new Date(p.createdAt).toLocaleDateString('vi-VN')}</span>

   419→                      {p.reviewNote && <p className="text-xs text-red-400">Lý do: {p.reviewNote}</p>}
   420→                    </div>
   421→                    <div className="flex items-center gap-1">
   422→                      {p.reviewStatus === 'PENDING' && (
   423→                        <>
