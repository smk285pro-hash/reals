     1→'use client'
     2→
     3→import { useState, useEffect } from 'react'
     4→import { useSession } from 'next-auth/react'
     5→import {
     6→  Users, Package, DollarSign, Clock, CheckCircle, XCircle,
     7→  Search, ArrowLeft, Shield, Eye, EyeOff,
     8→  Star, Trash2, UserCheck, TrendingUp, AlertCircle,
     9→  Flag, BarChart3, Ban, MessageSquare, ExternalLink,
    10→  ChevronDown, AlertTriangle, FileWarning
    11→} from 'lucide-react'
    12→import { Button } from '@/components/ui/button'
    13→import { Badge } from '@/components/ui/badge'
    14→import { Skeleton } from '@/components/ui/skeleton'
    15→import { useAppStore } from '@/stores'
    16→import { toast } from 'sonner'
    17→
    18→type Tab = 'overview' | 'products' | 'users' | 'reports' | 'analytics'
    19→
    20→interface Stats {
    21→  totalUsers: number; totalSellers: number; totalProducts: number
    22→  publishedProducts: number; pendingProducts: number; totalRevenue: number
    23→}
    24→
    25→interface AdminUser {
    26→  id: string; name: string | null; email: string; role: string
    27→  isSeller: boolean; avatar: string | null; createdAt: string
    28→  _count: { products: number; reviews: number }
    29→}
    30→
    31→interface AdminProduct {
    32→  id: string; title: string; price: number; isFree: boolean
    33→  published: boolean; reviewStatus: string; reviewNote: string | null
    34→  createdAt: string; seller: { name: string | null; email: string }
    35→}
    36→
    37→interface ReportItem {
    38→  id: string; type: string; targetId: string; reason: string
    39→  description: string | null; status: string; adminNote: string | null
    40→  createdAt: string; updatedAt: string
    41→  reporter: { id: string; name: string | null; email: string; avatar: string | null }
    42→  target: any
    43→}
    44→
    45→interface AnalyticsData {
    46→  range: string
    47→  usersByRole: { role: string; _count: number }[]
    48→  newUsers: number
    49→  productsByStatus: { reviewStatus: string; _count: number }[]
    50→  productsByCategory: { categorySlug: string; _count: number }[]
    51→  topSellers: any[]
    52→  topProducts: any[]
    53→  reportsByStatus: { status: string; _count: number }[]
    54→  reportsByType: { type: string; _count: number }[]
    55→  revenueByFormat: { format: string; _sum: { price: number | null; sales: number | null }; _count: number }[]
    56→  recentActivity: any[]
    57→}
    58→
    59→function StatCard({ icon: Icon, label, value, color }: any) {
    60→  return (
    61→    <div className="rounded-xl border border-[#303030] bg-[#1a1a1a] p-4">
    62→      <div className="flex items-center gap-3">
    63→        <div className={`rounded-lg p-2 ${color}`}>
    64→          <Icon className="h-5 w-5" />
    65→        </div>
    66→        <div>
    67→          <p className="text-xs text-[#888]">{label}</p>
    68→          <p className="text-2xl font-bold text-white">{value}</p>
    69→        </div>
    70→      </div>
    71→    </div>
    72→  )
    73→}
    74→
    75→const tabLabels: Record<Tab, string> = {
    76→  overview: 'Tổng quan',
    77→  products: 'Sản phẩm',
    78→  users: 'User',
    79→  reports: 'Báo cáo',
    80→  analytics: 'Analytics',
    81→}
    82→
    83→export default function AdminDashboard() {
    84→  const { setActiveCategory } = useAppStore()
    85→  const { status: authStatus, data: session } = useSession()
    86→  const [tab, setTab] = useState<Tab>('overview')
    87→  const [stats, setStats] = useState<Stats | null>(null)
    88→  const [users, setUsers] = useState<AdminUser[]>([])
    89→  const [products, setProducts] = useState<AdminProduct[]>([])
    90→  const [reports, setReports] = useState<ReportItem[]>([])
    91→  const [analytics, setAnalytics] = useState<AnalyticsData | null>(null)
    92→  const [loading, setLoading] = useState(true)
    93→  const [searchQ, setSearchQ] = useState('')
    94→  const [productFilter, setProductFilter] = useState('ALL')
    95→  const [reportFilter, setReportFilter] = useState('PENDING')
    96→  const [analyticsRange, setAnalyticsRange] = useState('30d')
    97→
    98→  const userRole = (session?.user as any)?.role
    99→
   100→  // Fetch data
   101→  const fetchStats = async () => {
   102→    try {
   103→      const res = await fetch('/api/admin/stats')
   104→      if (res.ok) setStats(await res.json())
   105→    } catch {}
   106→  }
   107→
   108→  const fetchUsers = async () => {
   109→    setLoading(true)
   110→    try {
   111→      const res = await fetch(`/api/admin/users?search=${searchQ}&limit=50`)
   112→      if (res.ok) { const data = await res.json(); setUsers(data.users || []) }
   113→    } catch {} finally { setLoading(false) }
   114→  }
   115→
   116→  const fetchProducts = async () => {
   117→    setLoading(true)
   118→    try {
   119→      const res = await fetch(`/api/admin/products?status=${productFilter}&search=${searchQ}&limit=50`)
   120→      if (res.ok) { const data = await res.json(); setProducts(data.products || []) }
   121→    } catch {} finally { setLoading(false) }
   122→  }
   123→
   124→  const fetchReports = async () => {
   125→    setLoading(true)
   126→    try {
   127→      const res = await fetch(`/api/admin/reports?status=${reportFilter}&limit=50`)
   128→      if (res.ok) { const data = await res.json(); setReports(data.reports || []) }
   129→    } catch {} finally { setLoading(false) }
   130→  }
   131→
   132→  const fetchAnalytics = async () => {
   133→    setLoading(true)
   134→    try {
   135→      const res = await fetch(`/api/admin/analytics?range=${analyticsRange}`)
   136→      if (res.ok) setAnalytics(await res.json())
   137→    } catch {} finally { setLoading(false) }
   138→  }
   139→
   140→  useEffect(() => { if (userRole === 'ADMIN') fetchStats() }, [userRole])
   141→  useEffect(() => { if (tab === 'users' && userRole === 'ADMIN') fetchUsers() }, [tab, userRole])
   142→  useEffect(() => { if (tab === 'products' && userRole === 'ADMIN') fetchProducts() }, [tab, productFilter, userRole])
   143→  useEffect(() => { if (tab === 'reports' && userRole === 'ADMIN') fetchReports() }, [tab, reportFilter, userRole])
   144→  useEffect(() => { if (tab === 'analytics' && userRole === 'ADMIN') fetchAnalytics() }, [tab, analyticsRange, userRole])
   145→
   146→  // Auth guard
   147→  if (authStatus === 'loading') {
   148→    return <div className="flex min-h-screen items-center justify-center bg-[#0f0f0f]"><Skeleton className="h-20 w-40" /></div>
   149→  }
   150→  if (authStatus === 'unauthenticated' || userRole !== 'ADMIN') {
   151→    return (
   152→      <div className="flex min-h-screen items-center justify-center bg-[#0f0f0f]">
   153→        <div className="flex flex-col items-center gap-4 rounded-2xl border border-[#303030] bg-[#181818] p-12">
   154→          <Shield className="h-16 w-16 text-red-500" />
   155→          <h2 className="text-xl font-bold text-white">403 - Access Denied</h2>
   156→          <p className="text-sm text-[#888]">Bạn không có quyền truy cập trang này</p>
   157→          <Button className="bg-[#f5a623] text-black hover:bg-[#e09515]" onClick={() => setActiveCategory('all')}>
   158→            Về trang chủ
   159→          </Button>
   160→        </div>
   161→      </div>
   162→    )
   163→  }
   164→
   165→  // Actions
   166→  const updateRole = async (id: string, role: string) => {
   167→    try {
   168→      const res = await fetch('/api/admin/users', {
   169→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   170→        body: JSON.stringify({ id, role }),
   171→      })
   172→      if (res.ok) { toast.success(`Đã cập nhật role thành ${role}`); fetchUsers() }
   173→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   174→    } catch { toast.error('Lỗi kết nối') }
   175→  }
   176→
   177→  const deleteUser = async (id: string) => {
   178→    if (!confirm('Xóa user này? Hành động không thể hoàn tác.')) return
   179→    try {
   180→      const res = await fetch('/api/admin/users', {
   181→        method: 'DELETE', headers: { 'Content-Type': 'application/json' },
   182→        body: JSON.stringify({ id }),
   183→      })
   184→      if (res.ok) { toast.success('Đã xóa user'); fetchUsers() }
   185→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   186→    } catch { toast.error('Lỗi kết nối') }
   187→  }
   188→
   189→  const reviewProduct = async (id: string, reviewStatus: string, reviewNote?: string) => {
   190→    try {
   191→      const res = await fetch('/api/admin/products', {
   192→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   193→        body: JSON.stringify({ id, reviewStatus, reviewNote }),
   194→      })
   195→      if (res.ok) { toast.success(reviewStatus === 'APPROVED' ? 'Đã duyệt' : 'Đã từ chối'); fetchProducts(); fetchStats() }
   196→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   197→    } catch { toast.error('Lỗi kết nối') }
   198→  }
   199→
   200→  const deleteProduct = async (id: string) => {
   201→    if (!confirm('Xóa sản phẩm này?')) return
   202→    try {
   203→      const res = await fetch('/api/admin/products', {
   204→        method: 'DELETE', headers: { 'Content-Type': 'application/json' },
   205→        body: JSON.stringify({ id }),
   206→      })
   207→      if (res.ok) { toast.success('Đã xóa'); fetchProducts(); fetchStats() }
   208→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   209→    } catch { toast.error('Lỗi kết nối') }
   210→  }
   211→
   212→  const togglePublish = async (id: string, published: boolean) => {
   213→    try {
   214→      const res = await fetch('/api/admin/products', {
   215→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   216→        body: JSON.stringify({ id, published }),
   217→      })
   218→      if (res.ok) { toast.success(published ? 'Đã ẩn' : 'Đã đăng'); fetchProducts() }
   219→      else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   220→    } catch { toast.error('Lỗi kết nối') }
   221→  }
   222→
   223→  const handleReport = async (id: string, status: string, adminNote?: string, actionType?: string) => {
   224→    try {
   225→      const res = await fetch('/api/admin/reports', {
   226→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   227→        body: JSON.stringify({ id, status, adminNote, actionType }),
   228→      })
   229→      if (res.ok) {
   230→        toast.success(status === 'ACTIONED' ? 'Đã xử lý' : status === 'DISMISSED' ? 'Đã bỏ qua' : 'Đã đánh dấu')
   231→        fetchReports(); fetchStats()
   232→      } else { const e = await res.json(); toast.error(e.error || 'Lỗi') }
   233→    } catch { toast.error('Lỗi kết nối') }
   234→  }
   235→
   236→  const roleColor: Record<string, string> = {
   237→    ADMIN: 'bg-red-500/20 text-red-400',
   238→    SELLER: 'bg-[#f5a623]/20 text-[#f5a623]',
   239→    USER: 'bg-[#3ea6ff]/20 text-[#3ea6ff]',
   240→  }
   241→
   242→  const reviewColor: Record<string, string> = {
   243→    PENDING: 'bg-yellow-500/20 text-yellow-400',
   244→    APPROVED: 'bg-[#3fb950]/20 text-[#3fb950]',
   245→    REJECTED: 'bg-red-500/20 text-red-400',
   246→  }
   247→
   248→  const reportStatusColor: Record<string, string> = {
   249→    PENDING: 'bg-yellow-500/20 text-yellow-400',
   250→    REVIEWED: 'bg-[#3ea6ff]/20 text-[#3ea6ff]',
   251→    DISMISSED: 'bg-[#888]/20 text-[#888]',
   252→    ACTIONED: 'bg-[#3fb950]/20 text-[#3fb950]',
   253→  }
   254→
   255→  const reasonLabel: Record<string, string> = {
   256→    SPAM: 'Spam',
   257→    INAPPROPRIATE: 'Không phù hợp',
   258→    COPYRIGHT: 'Vi phạm bản quyền',
   259→    OTHER: 'Khác',
   260→  }
   261→
   262→  const typeLabel: Record<string, string> = {
   263→    PRODUCT: 'Sản phẩm',
   264→    USER: 'User',
   265→    REVIEW: 'Review',
   266→  }
   267→
   268→  return (
   269→    <div className="min-h-screen bg-[#0f0f0f] text-[#f1f1f1]">
   270→      {/* Header */}
   271→      <div className="border-b border-[#303030] bg-[#0f0f0f] px-4 py-4 md:px-6">
   272→        <div className="mx-auto max-w-[1200px]">
   273→          <div className="flex items-center justify-between">
   274→            <div className="flex items-center gap-3">
   275→              <Button variant="ghost" size="icon" className="text-[#aaa] hover:bg-[#272727]" onClick={() => setActiveCategory('all')}>
   276→                <ArrowLeft className="h-5 w-5" />
   277→              </Button>
   278→              <div className="flex items-center gap-2">
   279→                <Shield className="h-6 w-6 text-red-500" />
   280→                <div>
   281→                  <h1 className="text-lg font-bold text-white">Admin Dashboard</h1>
   282→                  <p className="text-xs text-[#888]">Quản trị hệ thống RealS</p>
   283→                </div>
   284→              </div>
   285→            </div>
   286→            <div className="flex gap-1 rounded-lg bg-[#1f1f1f] p-1 overflow-x-auto">
   287→              {(['overview', 'products', 'users', 'reports', 'analytics'] as Tab[]).map(t => (
   288→                <button
   289→                  key={t}
   290→                  className={`whitespace-nowrap rounded-md px-3 py-1.5 text-sm font-medium transition-all ${
   291→                    tab === t ? 'bg-[#f5a623] text-black' : 'text-[#aaa] hover:text-white'
   292→                  }`}
   293→                  onClick={() => setTab(t)}
   294→                >
   295→                  {tabLabels[t]}
   296→                </button>
   297→              ))}
   298→            </div>
   299→          </div>
   300→        </div>
   301→      </div>
   302→
   303→      <div className="mx-auto max-w-[1200px] px-4 py-6 md:px-6">
   304→        {/* Overview Tab */}
   305→        {tab === 'overview' && stats && (
   306→          <div className="space-y-6">
   307→            <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   308→              <StatCard icon={Users} label="Tổng User" value={stats.totalUsers} color="bg-[#3ea6ff]/20 text-[#3ea6ff]" />
   309→              <StatCard icon={UserCheck} label="Seller" value={stats.totalSellers} color="bg-[#f5a623]/20 text-[#f5a623]" />
   310→              <StatCard icon={Package} label="Sản phẩm" value={stats.totalProducts} color="bg-[#a855f7]/20 text-[#a855f7]" />
   311→              <StatCard icon={CheckCircle} label="Đã đăng" value={stats.publishedProducts} color="bg-[#3fb950]/20 text-[#3fb950]" />
   312→              <StatCard icon={Clock} label="Chờ duyệt" value={stats.pendingProducts} color="bg-yellow-500/20 text-yellow-400" />
   313→              <StatCard icon={DollarSign} label="Tổng giá trị" value={`$${stats.totalRevenue.toFixed(2)}`} color="bg-[#3fb950]/20 text-[#3fb950]" />
   314→            </div>
   315→            {stats.pendingProducts > 0 && (
   316→              <div className="flex items-center gap-3 rounded-xl border border-yellow-500/30 bg-yellow-500/5 p-4">
   317→                <AlertCircle className="h-5 w-5 text-yellow-400" />
   318→                <span className="text-sm text-yellow-200">Có <strong>{stats.pendingProducts}</strong> sản phẩm đang chờ duyệt</span>
   319→                <Button size="sm" className="ml-auto bg-yellow-500 text-black hover:bg-yellow-600" onClick={() => { setProductFilter('PENDING'); setTab('products') }}>
   320→                  Xem ngay
   321→                </Button>
   322→              </div>
   323→            )}
   324→          </div>
   325→        )}
   326→
   327→        {/* Products Tab */}
   328→        {tab === 'products' && (
   329→          <div className="space-y-4">
   330→            <div className="flex flex-wrap items-center gap-3">
   331→              <div className="flex items-center gap-2 rounded-lg bg-[#1f1f1f] px-3 py-2">
   332→                <Search className="h-4 w-4 text-[#888]" />
   333→                <input value={searchQ} onChange={e => setSearchQ(e.target.value)} placeholder="Tìm sản phẩm..." className="bg-transparent text-sm text-white outline-none placeholder:text-[#666]" />
   334→              </div>
   335→              <div className="flex gap-1 rounded-lg bg-[#1f1f1f] p-1">
   336→                {['ALL', 'PENDING', 'APPROVED', 'REJECTED'].map(f => (
   337→                  <button key={f} className={`rounded-md px-3 py-1 text-xs font-medium ${productFilter === f ? 'bg-[#f5a623] text-black' : 'text-[#aaa] hover:text-white'}`} onClick={() => setProductFilter(f)}>
   338→                    {f === 'ALL' ? 'Tất cả' : f === 'PENDING' ? 'Chờ duyệt' : f === 'APPROVED' ? 'Đã duyệt' : 'Từ chối'}
   339→                  </button>
   340→                ))}
   341→              </div>
   342→            </div>
   343→            {loading ? (
   344→              <div className="space-y-2">{Array.from({length:5}).map((_,i) => <Skeleton key={i} className="h-16 w-full rounded-lg bg-[#1f1f1f]" />)}</div>
   345→            ) : products.length === 0 ? (
   346→              <div className="py-20 text-center text-[#888]">Không có sản phẩm</div>
   347→            ) : (
   348→              <div className="space-y-2">
   349→                {products.map(p => (
   350→                  <div key={p.id} className="flex items-center gap-3 rounded-xl border border-[#303030] bg-[#1a1a1a] p-4">
   351→                    <div className="flex-1 space-y-1">
   352→                      <div className="flex items-center gap-2">
   353→                        <h3 className="text-sm font-medium text-[#f1f1f1]">{p.title}</h3>
   354→                        <Badge className={reviewColor[p.reviewStatus] || ''}>{p.reviewStatus}</Badge>
   355→                        {p.featured && <Star className="h-3.5 w-3.5 fill-[#f5a623] text-[#f5a623]" />}
   356→                      </div>
   357→                      <div className="flex items-center gap-2 text-xs text-[#aaa]">
   358→                        <span>{p.seller.name || p.seller.email}</span>
   359→                        <span>•</span>
   360→                        <span className={p.isFree ? 'text-[#3fb950]' : 'text-[#f5a623]'}>{p.isFree ? 'FREE' : `$${p.price}`}</span>
   361→                        <span>•</span>
   362→                        <span>{new Date(p.createdAt).toLocaleDateString('vi-VN')}</span>
   363→                      </div>
   364→                      {p.reviewNote && <p className="text-xs text-red-400">Lý do: {p.reviewNote}</p>}
   365→                    </div>
   366→                    <div className="flex items-center gap-1">
   367→                      {p.reviewStatus === 'PENDING' && (
   368→                        <>
   369→                          <Button size="icon" variant="ghost" className="h-8 w-8 text-[#3fb950] hover:bg-[#3fb950]/10" onClick={() => reviewProduct(p.id, 'APPROVED')} title="Duyệt">
   370→                            <CheckCircle className="h-4 w-4" />
   371→                          </Button>
   372→                          <Button size="icon" variant="ghost" className="h-8 w-8 text-red-400 hover:bg-red-400/10" onClick={() => reviewProduct(p.id, 'REJECTED', 'Không đạt yêu cầu')} title="Từ chối">
   373→                            <XCircle className="h-4 w-4" />
   374→                          </Button>
   375→                        </>
   376→                      )}
   377→                      <Button size="icon" variant="ghost" className={`h-8 w-8 ${p.published ? 'text-[#3fb950]' : 'text-[#888]'} hover:bg-[#272727]`} onClick={() => togglePublish(p.id, p.published)} title={p.published ? 'Ẩn' : 'Đăng'}>
   378→                        {p.published ? <Eye className="h-4 w-4" /> : <EyeOff className="h-4 w-4" />}
   379→                      </Button>
   380→                      <Button size="icon" variant="ghost" className="h-8 w-8 text-red-400 hover:bg-red-400/10" onClick={() => deleteProduct(p.id)} title="Xóa">
   381→                        <Trash2 className="h-4 w-4" />
   382→                      </Button>
   383→                    </div>
   384→                  </div>
   385→                ))}
   386→              </div>
   387→            )}
   388→          </div>
   389→        )}
   390→
   391→        {/* Users Tab */}
   392→        {tab === 'users' && (
   393→          <div className="space-y-4">
   394→            <div className="flex items-center gap-2 rounded-lg bg-[#1f1f1f] px-3 py-2">
   395→              <Search className="h-4 w-4 text-[#888]" />
   396→              <input value={searchQ} onChange={e => setSearchQ(e.target.value)} onKeyDown={e => e.key === 'Enter' && fetchUsers()} placeholder="Tìm user..." className="bg-transparent text-sm text-white outline-none placeholder:text-[#666]" />
   397→            </div>
   398→            {loading ? (
   399→              <div className="space-y-2">{Array.from({length:5}).map((_,i) => <Skeleton key={i} className="h-16 w-full rounded-lg bg-[#1f1f1f]" />)}</div>
   400→            ) : users.length === 0 ? (
   401→              <div className="py-20 text-center text-[#888]">Không có user</div>
   402→            ) : (
   403→              <div className="space-y-2">
   404→                {users.map(u => (
   405→                  <div key={u.id} className="flex items-center gap-3 rounded-xl border border-[#303030] bg-[#1a1a1a] p-4">
   406→                    <div className="flex h-10 w-10 items-center justify-center rounded-full bg-[#303030] text-sm font-bold text-white">
   407→                      {(u.name || u.email)[0].toUpperCase()}
   408→                    </div>
   409→                    <div className="flex-1 space-y-1">
   410→                      <div className="flex items-center gap-2">
   411→                        <h3 className="text-sm font-medium text-[#f1f1f1]">{u.name || 'Chưa đặt tên'}</h3>
   412→                        <Badge className={roleColor[u.role] || ''}>{u.role}</Badge>
   413→                      </div>
   414→                      <div className="flex items-center gap-2 text-xs text-[#aaa]">
   415→                        <span>{u.email}</span>
   416→                        <span>•</span>
   417→                        <span>{u._count.products} sản phẩm</span>
   418→                        <span>•</span>
   419→                        <span>{u._count.reviews} review</span>
