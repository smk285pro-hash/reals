   118→  const [deleteConfirm, setDeleteConfirm] = useState<string | null>(null)
   119→
   120→  const fetchProducts = useCallback(async () => {

   122→    try {
   123→      const res = await fetch('/api/seller')
   124→      if (res.ok) {
   125→        const data = await res.json()

   130→        console.error('[fetchProducts] Error:', res.status, err)

   132→        setCategories([])
   133→        if (res.status === 401) {
   134→          toast.error('Vui lòng đăng nhập lại')

   136→          toast.error(err.error || 'Lỗi tải sản phẩm')
   137→        }
   138→      }
   139→    } catch (e) {

   141→      setProducts([])
   142→      setCategories([])
   143→      toast.error('Lỗi kết nối server')

   145→      setLoading(false)
   146→    }
   147→  }, [])
   148→
   149→  useEffect(() => { fetchProducts() }, [fetchProducts])
   150→

   168→            </p>
   169→          </div>
   170→          <Button
   171→            className="gap-2 rounded-lg bg-[#f5a623] px-8 text-black font-semibold hover:bg-[#e09515]"

   178→            variant="ghost"
   179→            className="text-[#888] hover:text-[#f1f1f1]"

   192→  const rejectedCount = products.filter((p) => p.reviewStatus === 'REJECTED').length
   193→

   212→        toast.success(product.published ? 'Đã ẩn sản phẩm' : 'Đã đăng sản phẩm')

   215→        const err = await res.json().catch(() => ({}))

   239→

   298→    }

   301→      let success = false
   302→      if (editingId) {
   303→        const res = await fetch(`/api/products/${editingId}`, {
   304→          method: 'PUT',
   305→          headers: { 'Content-Type': 'application/json' },
   306→          body: JSON.stringify(form),
   307→        })
   308→        if (res.ok) {
   309→          toast.success('Đã cập nhật sản phẩm')
   310→          success = true
   311→        } else {
   312→          const err = await res.json().catch(() => ({}))
   313→          toast.error(err.error || `Lỗi cập nhật (${res.status})`)
   314→        }
   315→      } else {
   316→        // New products always start as PENDING for review
   317→        const res = await fetch('/api/products', {
   318→          method: 'POST',
   319→          headers: { 'Content-Type': 'application/json' },
   320→          body: JSON.stringify({ ...form, reviewStatus: 'PENDING' }),
   321→        })
   322→        if (res.ok) {
   323→          toast.success('Đã đăng sản phẩm mới — đang chờ duyệt')
   324→          success = true
   325→        } else {
   326→          const err = await res.json().catch(() => ({}))
   327→          toast.error(err.error || `Lỗi đăng sản phẩm (${res.status})`)
   328→        }
   329→      }
   330→      if (success) {
   331→        setShowForm(false)
   332→        await fetchProducts()
   333→      }
   334→    } catch (e) {
   335→      console.error('[handleSave] Error:', e)
   336→      toast.error('Lỗi kết nối server')
   337→    } finally {
   338→      setSaving(false)
   339→    }
   340→  }
   341→
   342→  const thumbnailPresets = [

   344→    'https://images.unsplash.com/photo-1558618660-7c0c3b1a4e93?w=640&h=360&fit=crop',

   348→    'https://images.unsplash.com/photo-1493225452364-bab4a9fcd274?w=640&h=360&fit=crop',

   350→
   351→  const statusFilterOptions: { value: StatusFilter; label: string; count: number }[] = [

   354→    { value: 'APPROVED', label: 'Đã duyệt', count: approvedCount },
   355→    { value: 'REJECTED', label: 'Bị từ chối', count: rejectedCount },
   356→  ]
   357→
   358→  return (
   359→    <div className="min-h-screen bg-[#0f0f0f] text-[#f1f1f1]">
   360→      {/* Header */}
   361→      <div className="border-b border-[#303030] bg-[#0f0f0f] px-4 py-4 md:px-6">
   362→        <div className="mx-auto max-w-[1200px]">
   363→          <div className="flex items-center justify-between">
   364→            <div className="flex items-center gap-3">

   367→                size="icon"

   383→                  onChange={(e) => setSearchQ(e.target.value)}
   384→                  placeholder="Tìm sản phẩm..."
   385→                  className="bg-transparent text-sm text-white outline-none placeholder:text-[#666]"

   388→              <Button

   398→          {/* Stats bar */}

   459→        {/* Product Form Modal */}
